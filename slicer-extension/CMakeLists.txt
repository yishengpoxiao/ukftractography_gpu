cmake_minimum_required(VERSION 3.22)
project(UKFGPU)
set(EXTENSION_HOMEPAGE "https://github.com/pnlbwh/ukftractography")
set(EXTENSION_CATEGORY "Diffusion.Tractography")
set(EXTENSION_CONTRIBUTORS "UKF GPU contributors; original UKFTractography authors")
set(EXTENSION_DESCRIPTION "Experimental GPU UKF tractography with Simple, Full Tensor, FreeWater and NODDI models. Uses an isolated Linux CUDA backend.")
set(EXTENSION_ICONURL "https://raw.githubusercontent.com/pnlbwh/ukftractography/master/UKF_icon.png")
set(EXTENSION_SCREENSHOTURLS "")
set(EXTENSION_DEPENDS "NA")
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
add_subdirectory(UKFGPU)
include(${Slicer_EXTENSION_GENERATE_CONFIG})
include(${Slicer_EXTENSION_CPACK})
